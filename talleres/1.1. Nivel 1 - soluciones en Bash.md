# Soluciones en Bash para 20 ejercicios (STDIN/STDOUT)

A continuación se muestran soluciones en **Bash** para los 20 ejercicios básicos, usando únicamente **entrada estándar** y **salida estándar**.

---

## 1. Suma de dos números
```bash
read a
read b
echo $((a + b))
```

## 2. Contador de vocales
```bash
read str
str=$(echo "$str" | tr 'A-Z' 'a-z')
count=0
for ((i=0; i<${#str}; i++)); do
    c=${str:$i:1}
    [[ $c =~ [aeiou] ]] && count=$((count+1))
done
echo $count
```

## 3. Números pares e impares
```bash
read n
for ((i=1; i<=n; i++)); do
    if (( i % 2 == 0 )); then
        echo "$i par"
    else
        echo "$i impar"
    fi
done
```

## 4. Tabla de multiplicar
```bash
read n
for ((i=1; i<=10; i++)); do
    echo "$n x $i = $((n*i))"
done
```

## 5. Calculadora básica
```bash
read a
read b
read op

case $op in
    +) echo $((a + b)) ;;
    -) echo $((a - b)) ;;
    \*) echo $((a * b)) ;;
    /) [[ $b -ne 0 ]] && echo $((a / b)) || echo "Error: división por cero" ;;
    *) echo "Operación no válida" ;;
esac
```

## 6. Conversor de temperatura
```bash
read val
read unit
unit=${unit^^}

if [[ $unit == "C" ]]; then
    echo "scale=2; ($val * 9/5) + 32" | bc
elif [[ $unit == "F" ]]; then
    echo "scale=2; ($val - 32) * 5/9" | bc
else
    echo "Unidad no válida"
fi
```

## 7. Inversión de cadena
```bash
read str
echo "$str" | rev
```

## 8. Determinar si es primo
```bash
read n
if (( n < 2 )); then echo "No primo"; exit; fi

for ((i=2; i*i<=n; i++)); do
    if (( n % i == 0 )); then
        echo "No primo"
        exit
    fi
done
echo "Primo"
```

## 9. Media de una lista
```bash
read -a nums
sum=0
for n in "${nums[@]}"; do
    sum=$((sum + n))
done
echo $((sum / ${#nums[@]}))
```

## 10. Frecuencia de caracteres
```bash
read str
declare -A freq
for ((i=0; i<${#str}; i++)); do
    c=${str:$i:1}
    freq[$c]=$((freq[$c]+1))
done
for k in "${!freq[@]}"; do
    echo "$k : ${freq[$k]}"
done
```

## 11. Máximo y mínimo
```bash
read -a nums
max=${nums[0]}
min=${nums[0]}
for n in "${nums[@]}"; do
    (( n > max )) && max=$n
    (( n < min )) && min=$n
done
echo "Max: $max"
echo "Min: $min"
```

## 12. Contador de palabras
```bash
read str
echo "$str" | wc -w
```

## 13. Conversor de segundos
```bash
read s
h=$((s/3600))
m=$(((s%3600)/60))
sec=$((s%60))
echo "$h h $m m $sec s"
```

## 14. Número aleatorio entre límites
```bash
read min
read max
echo $(( RANDOM % (max - min + 1) + min ))
```

## 15. Cifrado César
```bash
read txt
read shift
res=""

for ((i=0; i<${#txt}; i++)); do
    c=${txt:$i:1}
    if [[ $c =~ [A-Za-z] ]]; then
        if [[ $c =~ [A-Z] ]]; then base=65; else base=97; fi
        ord=$(printf "%d" "'$c")
        new=$(( (ord - base + shift) % 26 + base ))
        res+=$(printf \$(printf '%03o' $new))
    else
        res+=$c
    fi
done
echo "$res"
```

## 16. Ordenar números
```bash
read -a nums
printf "%s
" "${nums[@]}" | sort -n | tr '
' ' '
echo
```

## 17. Validación de contraseña
```bash
read pwd
if [[ ${#pwd} -ge 8 && $pwd =~ [A-Z] && $pwd =~ [a-z] && $pwd =~ [0-9] ]]; then
    echo "Válida"
else
    echo "Inválida"
fi
```

## 18. Contador de líneas
```bash
count=0
while true; do
    read line
    [[ -z "$line" ]] && break
    count=$((count+1))
done
echo $count
```

## 19. Simulación de dado
```bash
echo "Pulsa Enter para lanzar (vacío para salir)"
while true; do
    read t
    [[ -z "$t" ]] && break
    echo $(( RANDOM % 6 + 1 ))
done
```

## 20. Decimal a binario
```bash
read n
echo "obase=2; $n" | bc
```
